\begin{rem}[Working Treatment of Operator Domains]{operator-domain-convention}
  Fix a \xref[Hilbert space]{ch03::def:hilbert-space} $\mathcal H$. Throughout this chapter, formally write $A : \mathcal H \to \mathcal H$ for an operator $A \in \OP$.
  This notation is literal for finite-dimensional matrix operators. For differential operators, it suppresses domain restrictions: all expressions are understood on a suitable common domain where the indicated operations are defined.
  Detailed operator domains and self-adjointness will be addressed when revising Chapters 2 and 3 rigorously
\end{rem}
\endgroup

\begin{dfn}[Unperturbed Hamiltonian]{unperturbed-hamiltonian}
  An \emph{unperturbed Hamiltonian} $\hat H_0 \in \OP$ is a reference time-independent \missingprereq{Hamiltonian, quantum-mechanics} whose eigenstates and corresponding energies are taken as known for the perturbation calculation
\end{dfn}

\begin{dfn}[Perturbation Operator]{perturbation}
  Given an \xref[unperturbed Hamiltonian]{def:unperturbed-hamiltonian} $\hat H_0$ and the physical time-independent Hamiltonian $\hat H \in \OP$ on the same state space $\mathcal H$, the \emph{perturbation operator} is their difference:
  \[
    \hat H' \defeq \hat H - \hat H_0
  \]
\end{dfn}

\begin{dfn}[Perturbation Family]{perturbation-family}
  Fix an \xref[unperturbed Hamiltonian]{def:unperturbed-hamiltonian} $\hat H_0$ and its \xref[perturbation operator]{def:perturbation} $\hat H'$. Let $I \subseteq \mathbb R$ be an \xref[interval]{real-analysis::lec02::def:interval} containing $0$ and $1$.
  The \emph{perturbation family} determined by these operators is the function
  \[
    \hat H_{\mathrm{fam}} : I \to \OP \defeq \lambda \map \hat H_0 + \lambda \cdot \hat H'.
  \]
  Require each $\hat H_{\mathrm{fam}}(\lambda)$ to be \xref[Hermitian]{ch03::def:hermitian-operator}, with domain restrictions suppressed as in the \xref[working treatment of operator domains]{rem:operator-domain-convention}
\end{dfn}

\begin{conv}[Perturbation Terminology]{perturbation-terminology}
  For the \xref[perturbation family]{def:perturbation-family}, use the following terminology:
  \begin{description}[nosep]
    \item[$\hat H$:] the full physical Hamiltonian, also called the \emph{perturbed Hamiltonian}.
    \item[$\hat H_0$:] the \xref[unperturbed Hamiltonian]{def:unperturbed-hamiltonian}, whose eigenstates and energies are taken as known.
    \item[$\hat H'$:] the \xref[perturbation operator]{def:perturbation}, equal to $\hat H - \hat H_0$.
    \item[$\lambda$:] the dimensionless \emph{perturbation parameter}, treated as small when deriving corrections. The value $0$ gives the reference Hamiltonian, and $1$ gives the full physical Hamiltonian
  \end{description}
\end{conv}

\begin{rem}[Prime on $\hat{H}'$ Is a Label, Not a Derivative]{perturbation-prime-notation}
\end{rem}

\begin{dfn}[Reference Eigenpair]{reference-eigenpair}
  A \emph{reference eigenpair} of an \xref[unperturbed Hamiltonian]{def:unperturbed-hamiltonian} $\hat H_0$ is a known pair $(\psi_n^{\porder{0}}, E_n^{\porder{0}}) \in \mathcal H \times \mathbb R$ satisfying
  \[
    \psi_n^{\porder{0}} \ne 0,
    \qquad
    \hat H_0\bigl(\psi_n^{\porder{0}}\bigr) = E_n^{\porder{0}} \cdot \psi_n^{\porder{0}}
  \]
\end{dfn}

\begin{dfn}[Eigenvector Branch]{eigenvector-branch}
  Given a \xref[perturbation family]{def:perturbation-family} and a \xref[reference eigenpair]{def:reference-eigenpair} $(\psi_n^{\porder{0}}, E_n^{\porder{0}})$, an \emph{eigenvector branch} is a continuous function $\phi : I \to \mathcal H$ with $\phi(0) = \psi_n^{\porder{0}}$ and, for every $\lambda \in I$,
  \[
    \phi(\lambda) \ne 0,
    \qquad
    \exists e \in \mathbb R,\quad
    \hat H_{\mathrm{fam}}(\lambda)\bigl(\phi(\lambda)\bigr)
    = e \cdot \phi(\lambda)
  \]
\end{dfn}

\begin{dfn}[Eigenvalue Branch]{eigenvalue-branch}
  For a chosen \xref[eigenvector branch]{def:eigenvector-branch} $\phi_n$, its \emph{eigenvalue branch} is the function $\varepsilon_n : I \to \mathbb R$ satisfying
  \[
    \hat H_{\mathrm{fam}}(\lambda)\bigl(\phi_n(\lambda)\bigr)
    = \varepsilon_n(\lambda) \cdot \phi_n(\lambda)
    \qquad (\lambda \in I)
  \]
\end{dfn}

\begin{rem}[Choosing a Branch]{eigenvector-branch-choice}
  The \xref[eigenvector-branch definition]{def:eigenvector-branch} specifies possible functions without asserting existence or uniqueness. If one exists, fix a choice $\phi_n$; the index $n$ labels its reference eigenpair.
  For this choice, each $\varepsilon_n(\lambda)$ exists and is unique: two candidate eigenvalues $e,f$ satisfy $(e-f) \cdot \phi_n(\lambda) = 0$, so $e=f$. In particular, $\varepsilon_n(0) = E_n^{\porder{0}}$
\end{rem}

\begin{conv}[State and Energy Notation]{eigenpair-notation}
  \[
    \psi_n^{\porder{0}} = \phi_n(0),
    \qquad
    E_n^{\porder{0}} = \varepsilon_n(0).
  \]
  \[
    \psi_n^{\mathrm{phys}} \defeq \phi_n(1),
    \qquad
    E_n^{\mathrm{phys}} \defeq \varepsilon_n(1)
  \]
\end{conv}

\begin{dfn}[State Coefficient Sequence]{perturbative-expansion}
  \[
    \psi_n : \mathbb N_0 \to \mathcal H \defeq k \map \psi_n^{\porder{k}}
  \]
\end{dfn}

\begin{dfn}[Energy Coefficient Sequence]{energy-perturbative-expansion}
  \[
    E_n : \mathbb N_0 \to \mathbb R \defeq k \map E_n^{\porder{k}}
  \]
\end{dfn}

\begin{hypo}[Existence of a Perturbative Expansion]{perturbative-expansion-existence}
  For a chosen \xref[eigenvector branch]{def:eigenvector-branch} $\phi_n$ and its \xref[eigenvalue branch]{def:eigenvalue-branch} $\varepsilon_n$, assume there exist a \xref[state coefficient sequence]{def:perturbative-expansion} $\psi_n$, an \xref[energy coefficient sequence]{def:energy-perturbative-expansion} $E_n$, and a neighborhood $J$ of $0$ in $I$ such that the following local function rules hold for $\lambda \in J$:
  \[
    \phi_n : I \to \mathcal H,
    \qquad \lambda \map \sum_{0}^{\infty}\left(k \map \lambda^k \cdot \psi_n(k)\right).
  \]
  \[
    \varepsilon_n : I \to \mathbb R,
    \qquad \lambda \map \sum_{0}^{\infty}\left(k \map \lambda^k \cdot E_n(k)\right).
  \]
  Moreover, for each $A \in \{\hat H_0,\hat H'\}$, all indicated operator applications are defined and
  \[
    A\circ\phi_n : I \to \mathcal H,
    \qquad \lambda \map \sum_{0}^{\infty}\left(k \map \lambda^k \cdot A\bigl(\psi_n(k)\bigr)\right).
  \]
  For every $\lambda \in J$, all vector series above converge in the norm of $\mathcal H$, and the scalar series converges in $\mathbb R$
\end{hypo}

\begin{rem}[Scope of the Expansion Hypothesis]{expansion-convergence}
  The \xref[expansion hypothesis]{hypo:perturbative-expansion-existence} assumes convergence near $\lambda = 0$; matching coefficients alone does not prove convergence. Validity at the physical endpoint $\lambda = 1$ requires separate justification
\end{rem}

\begin{thm}[Perturbation Equation (Summation Form)]{perturbation-equation}
  Under the \xref[expansion hypothesis]{hypo:perturbative-expansion-existence}, for $\lambda \in I$ sufficiently close to $0$,
  \[
    \begin{aligned}
      &\left(\hat H_0 + \lambda \cdot \hat H'\right)
      \left(\sum_{0}^{\infty}\left(k \map \lambda^k \cdot \psi_n(k)\right)\right) \\
      &\quad =
      \left(\sum_{0}^{\infty}\left(j \map \lambda^j \cdot E_n(j)\right)\right)
      \cdot
      \left(\sum_{0}^{\infty}\left(k \map \lambda^k \cdot \psi_n(k)\right)\right)
    \end{aligned}
  \]
\end{thm}

\begin{proof}
  Substitute $\hat H_{\mathrm{fam}}(\lambda) = \hat H_0 + \lambda \cdot \hat H'$ and the \xref[assumed expansions]{hypo:perturbative-expansion-existence} into the TISE
\end{proof}

\begin{conv}[Perturbation Order Notation]{order-k-correction}
  Parenthesized superscripts denote perturbative order:
  \[
    \psi_n^{\porder{k}} \defeq \psi_n(k),
    \qquad
    E_n^{\porder{k}} \defeq E_n(k).
  \]
  For $k \geq 1$, these are the \emph{$k$-th-order state and energy correction coefficients}. The zeroth-order terms form the reference eigenpair:
  \[
    \psi_n^{\porder{0}} = \phi_n(0),
    \qquad
    E_n^{\porder{0}} = \varepsilon_n(0)
  \]
\end{conv}

\begin{dfn}[Order-by-Order Perturbation System]{order-matched-system}
  For coefficient sequences $\psi_n,E_n$, the \emph{order-by-order perturbation system} consists of
  \[
    \hat H_0\bigl(\psi_n(0)\bigr) = E_n(0) \cdot \psi_n(0)
  \]
  and, for every $k \in \mathbb N_{>0}$,
  \[
    \hat H_0\bigl(\psi_n(k)\bigr) + \hat H'\bigl(\psi_n(k-1)\bigr)
    = \sum_{0}^{k}\left(j \map E_n(j) \cdot \psi_n(k-j)\right)
  \]
\end{dfn}

\begin{thm}[Equivalence of Perturbation Equations]{order-matched-equations}

\end{thm}

\begin{rem}[Recovering the Summation Equation]{recovering-summation-equation}
  Multiply the order-$k$ equation by $\lambda^k$ and sum its left-hand and right-hand sides separately over all orders. Here $k$ is the correction order, while $n$ labels the reference state. The shifted perturbation sum becomes
  \[
    \sum_{1}^{\infty}\left(k \map \lambda^k \cdot \hat H'\bigl(\psi_n(k-1)\bigr)\right)
    = \lambda \cdot \hat H'\bigl(\phi_n(\lambda)\bigr),
  \]
  and the summed right-hand side becomes $\varepsilon_n(\lambda) \cdot \phi_n(\lambda)$. Thus
  \[
    \left(\hat H_0 + \lambda \cdot \hat H'\right)\bigl(\phi_n(\lambda)\bigr)
    = \varepsilon_n(\lambda) \cdot \phi_n(\lambda)
  \]
\end{rem}

\begin{cor}[Zeroth-, First-, and Second-Order Equations]{equations-through-second-order}
  \[
    \begin{aligned}
      \hat H_0\bigl(\psi_n^{\porder{0}}\bigr)
      &= E_n^{\porder{0}} \cdot \psi_n^{\porder{0}}, \\
      \hat H_0\bigl(\psi_n^{\porder{1}}\bigr) + \hat H'\bigl(\psi_n^{\porder{0}}\bigr)
      &= E_n^{\porder{0}} \cdot \psi_n^{\porder{1}} + E_n^{\porder{1}} \cdot \psi_n^{\porder{0}}, \\
      \hat H_0\bigl(\psi_n^{\porder{2}}\bigr) + \hat H'\bigl(\psi_n^{\porder{1}}\bigr)
      &= E_n^{\porder{0}} \cdot \psi_n^{\porder{2}} + E_n^{\porder{1}} \cdot \psi_n^{\porder{1}}
      + E_n^{\porder{2}} \cdot \psi_n^{\porder{0}}
    \end{aligned}
  \]
\end{cor}

\begin{proof}
  Take orders $0,1,2$ in the \xref[order-by-order system]{def:order-matched-system}
\end{proof}

\begin{dfn}[$k$-th-Order Approximation]{perturbative-approximation}
  Let $\psi_n^{\porder{k}}$ and $E_n^{\porder{k}}$ be the solution for the \xref[order-by-order system]{def:order-matched-system} at order $k$

  The \emph{$k$-th-order perturbative approximation} to the eigenpair $(\phi_n(\lambda),\varepsilon_n(\lambda))$ is
  \[
    \phi_n^{\porder{k}} \defeq \sum_{0}^{k}\left(i \map \psi_n^{\porder{i}}\right),
    \qquad
    \varepsilon_n^{\porder{k}} \defeq \sum_{0}^{k}\left(i \map E_n^{\porder{i}}\right)
  \] 
\end{dfn}

\begin{rem}[Validity of the Approximation]{perturbative-validity}
  Since the equation works at $\lambda$ sufficiently close to $0$, the $k$-th-order approximation shall be significant small when $k$ is bigger
\end{rem}

\section{First-Order Perturbation Theory} 

\begin{thm}[First Order Energy Correction]{first-order-energy-correction}
  Under the \xref[expansion hypothesis]{hypo:perturbative-expansion-existence}, let the \xref[reference eigenstate]{def:reference-eigenpair} be normalized: $\braket{\psi_n^{\porder{0}}|\psi_n^{\porder{0}}} = 1$. Then
  \[
    E_n^{\porder{1}} = \bra{\psi_n^{\porder{0}}}\hat H'\ket{\psi_n^{\porder{0}}}
  \]
\end{thm} 

\begin{proof}
  According to the \xref[first order equation]{cor:equations-through-second-order}, we have 
  \[\hat H_0 \ket{\psi_n^{\porder{1}}} + \hat H' \ket{\psi_n^{\porder{0}}} = E_n^{\porder{0}} \ket{\psi_n^{\porder{1}}} + E_n^{\porder{1}} \ket{\psi_n^{\porder{0}}} \]

  Since $\psi_n^{\porder{0}}$ is a non-zero eigenvector, we can multiply its bra vector to both sides, ie 
  \[\bra{\psi_n^{\porder{0}}} \hat H_0 \ket{\psi_n^{\porder{1}}} + \bra{\psi_n^{\porder{0}}} \hat H' \ket{\psi_n^{\porder{0}}} = \bra{\psi_n^{\porder{0}}} E_n^{\porder{0}} \ket{\psi_n^{\porder{1}}} + \bra{\psi_n^{\porder{0}}} E_n^{\porder{1}} \ket{\psi_n^{\porder{0}}} \]

  Notice $E_n^{\porder{0}}$ is an real number, and according to the normalization axiom, 
  \[\bra{\psi_n^{\porder{0}}} E_n^{\porder{1}} \ket{\psi_n^{\porder{0}}} = E_n^{\porder{1}} \braket{\psi_n^{\porder{0}} \mid \psi_n^{\porder{0}}} = E_n^{\porder{1}} \] 

  According to the TISE, $H \ket{\psi} = E \ket{\psi}$, we subsitute that relationship into $\bra{\psi_n^{\porder{0}}} \hat H_0$ 
  \[\bra{\psi_n^{\porder{0}}} \hat H_0 \ket{\psi_n^{\porder{1}}} = \bra{\psi_n^{\porder{0}}} \hat E_0 \ket{\psi_n^{\porder{1}}} = E_0 \braket{\psi_n^{\porder{0}} \mid \psi_n^{\porder{1}}} \]

  Subsitute back, we can get 
  \[E_0 \braket{\psi_n^{\porder{0}} \mid \psi_n^{\porder{1}}} + \bra{\psi_n^{\porder{0}}} \hat H' \ket{\psi_n^{\porder{0}}} = \bra{\psi_n^{\porder{0}}} E_n^{\porder{0}} \ket{\psi_n^{\porder{1}}} + E_n^{\porder{1}} \]

  Notice that $E_0 \braket{\psi_n^{\porder{0}} \mid \psi_n^{\porder{1}}} = \bra{\psi_n^{\porder{0}}} E_n^{\porder{0}} \ket{\psi_n^{\porder{1}}} $, cancel out the term, 
  \[E_n^{\porder{1}}  = \bra{\psi_n^{\porder{0}}} \hat H' \ket{\psi_n^{\porder{0}}} \] 
\end{proof}

\begin{thm}[First Order State Correction]{first-order-state-correction}
  Under the \xref[expansion hypothesis]{hypo:perturbative-expansion-existence}, suppose the reference eigenstates are orthonormal and $\psi_n^{\porder{1}}$ admits a norm-convergent expansion in them. Let $E_n^{\porder{0}}$ be nondegenerate, and choose $\braket{\psi_n^{\porder{0}}|\psi_n^{\porder{1}}}=0$. Then
  \[
    \psi_n^{\porder{1}}
    = \sum_{m\ne n}\left(m \map
      \frac{\bra{\psi_m^{\porder{0}}}\hat H'\ket{\psi_n^{\porder{0}}}}
      {E_n^{\porder{0}}-E_m^{\porder{0}}}
      \cdot\psi_m^{\porder{0}}\right),
  \]
  where the sum runs over $m\in\mathbb N_1$ with $m\ne n$
\end{thm} 

\begin{proof}
  According to the \xref[first order equation]{cor:equations-through-second-order}, and multiply $\bra{\psi_m^{\porder{0}}}$ (the bra of a reference eigenstate) on both sides, we have
  \[
    \begin{aligned}
      &\bra{\psi_m^{\porder{0}}}\hat H_0\ket{\psi_n^{\porder{1}}}
      + \bra{\psi_m^{\porder{0}}}\hat H'\ket{\psi_n^{\porder{0}}} \\
      &\quad = \bra{\psi_m^{\porder{0}}}E_n^{\porder{0}}\ket{\psi_n^{\porder{1}}}
      + \bra{\psi_m^{\porder{0}}}E_n^{\porder{1}}\ket{\psi_n^{\porder{0}}}
    \end{aligned}
  \]

  Rearranging the equation, we can get 
  \[
      \bra{\psi_m^{\porder{0}}}\hat H_0\ket{\psi_n^{\porder{1}}}
      - E_n^{\porder{0}}\cdot\braket{\psi_m^{\porder{0}}|\psi_n^{\porder{1}}} 
       = E_n^{\porder{1}}\cdot\braket{\psi_m^{\porder{0}}|\psi_n^{\porder{0}}}
      - \bra{\psi_m^{\porder{0}}}\hat H'\ket{\psi_n^{\porder{0}}}
  \]

  Apply Hermiticity and the reference eigenvalue equation, we get 
  \[\bra{\psi_m^{\porder{0}}}\hat H_0\ket{\psi_n^{\porder{1}}} = \braket{ \hat H_0 \psi_m^{\porder{0}} \mid \psi_n^{\porder{1}}} = E_m^{\porder{0}}\braket{\psi_m^{\porder{0}} \mid \psi_n^{\porder{1}}}\] 

  Also, according \missingprereq{orthonormality}, 
  \[E_n^{\porder{1}}\cdot\braket{\psi_m^{\porder{0}}|\psi_n^{\porder{0}}} = E_n^{\porder{1}}\cdot\delta_{n, m}\]

  Let $\{\psi_m^{\porder{0}} : m \in \mathbb N_1\}$ be linearly independent reference solutions. For fixed $n$, assume there exists a coefficient function $c_n : \mathbb N_1 \to \mathbb C$ such that
  \[
    \psi_n^{\porder{1}}
    = \sum^{\infty}\left(m \map c_n(m)\cdot\psi_m^{\porder{0}}\right) 
  \]
  with convergence in the norm of $\mathcal H$. Which means that 
  \[c_m \defeq c_n(m) = \braket{\psi_m^{\porder{0}} \mid \psi_n^{\porder{1}}}\]

  Plug back in, we get 
  \[
  E_m^{\porder{0}} \cdot c_m 
      - E_n^{\porder{0}}\cdot c_m 
       = E_n^{\porder{1}}\cdot \delta_{n, m}
      - \bra{\psi_m^{\porder{0}}}\hat H'\ket{\psi_n^{\porder{0}}}\] 

  Choose $c_n(n)=0$. \todo: justify this normalization and phase choice

    For $m\ne n$ with $E_m^{\porder{0}}\ne E_n^{\porder{0}}$, solve for $c_m$, we have
    \[c_m = \frac{E_n^{\porder{1}}\cdot \delta_{n, m} - \bra{\psi_m^{\porder{0}}}\hat H'\ket{\psi_n^{\porder{0}}}}{E_m^{\porder{0}} - E_n^{\porder{0}}} \] 

  Since $m\ne n$, $\delta_{n,m}=0$. Hence
  \[
    c_m
    = -\frac{\bra{\psi_m^{\porder{0}}}\hat H'\ket{\psi_n^{\porder{0}}}}
      {E_m^{\porder{0}}-E_n^{\porder{0}}}
    = \frac{\bra{\psi_m^{\porder{0}}}\hat H'\ket{\psi_n^{\porder{0}}}}
      {E_n^{\porder{0}}-E_m^{\porder{0}}}.
  \]
  Substituting these coefficients into the expansion and using $c_n(n)=0$ gives
  \[
    \psi_n^{\porder{1}}
    = \sum_{m\ne n}\left(m \map
      \frac{\bra{\psi_m^{\porder{0}}}\hat H'\ket{\psi_n^{\porder{0}}}}
      {E_n^{\porder{0}}-E_m^{\porder{0}}}
      \cdot\psi_m^{\porder{0}}\right)
  \]
\end{proof}

\begin{rem}[Normalization and Phase]{eigenpair-choices}
  An \xref[eigenvector branch]{def:eigenvector-branch} is not unique: for any continuous function $c : I \to \mathbb C \setminus \{0\}$ with $c(0) = 1$, the branch
  \[
    \widetilde\phi_n : I \to \mathcal H \defeq \lambda \map c(\lambda) \cdot \phi_n(\lambda)
  \]
  has the same starting vector and \xref[eigenvalue branch]{def:eigenvalue-branch} by linearity.
  Unit normalization leaves a parameter-dependent phase undetermined
\end{rem}

\section{Degenerate Perturbation Theory}

\begin{dfn}[Good Basis]{good-basis}
  Let the degenerate eigenspace of the \xref[unperturbed Hamiltonian]{def:unperturbed-hamiltonian} be
  \[
    V_{E_0} \defeq \{\psi\in\mathcal H : \hat H_0(\psi)=E_0\cdot\psi\},
    \qquad \dim V_{E_0}=d\geq2.
  \]
  A \emph{good basis} is an orthonormal basis $B=\{\chi_r^{\porder{0}} : r\in\{1,\ldots,d\}\}$ of $V_{E_0}$, where for each $r$ there exists an \xref[eigenvector branch]{def:eigenvector-branch} $\phi_r$, satisfying the \xref[expansion hypothesis]{hypo:perturbative-expansion-existence}, with corresponding \xref[eigenvalue branch]{def:eigenvalue-branch} $\varepsilon_r$ such that
  \[
    \phi_r(0)=\chi_r^{\porder{0}},
    \qquad \varepsilon_r(0)=E_0
  \]
\end{dfn}

\begin{dfn}[Good States]{good-states}
  The \emph{good states} are the states in a \xref[good basis]{def:good-basis} $B$, ie, each $\chi_r^{\porder{0}}\in B$ is a good state
\end{dfn}

\begin{cor}[Common Unperturbed Energy]{good-states-common-energy}
  Given a \xref[good basis]{def:good-basis} $B$ of $V_{E_0}$ and a coefficient function $c : \{1,\ldots,d\}\to\mathbb C$, let
  \[
    \psi \defeq \sum_1^d\left(r \map c(r)\cdot\chi_r^{\porder{0}}\right).
  \]
  If $\psi\ne0$, it is an eigenstate of $\hat H_0$ with energy $E_0$:
  \[
    \hat H_0(\psi)=E_0\cdot\psi.
  \]
  Thus all good states and their nonzero linear combinations have the same unperturbed energy
\end{cor}

\begin{proof}
  Since each $\chi_r^{\porder{0}}\in V_{E_0}$, we have $\hat H_0(\chi_r^{\porder{0}})=E_0\cdot\chi_r^{\porder{0}}$. By linearity,
  \[
    \begin{aligned}
      \hat H_0(\psi)
      &= \sum_1^d\left(r \map c(r)\cdot\hat H_0(\chi_r^{\porder{0}})\right) \\
      &= E_0\cdot\sum_1^d\left(r \map c(r)\cdot\chi_r^{\porder{0}}\right)
      = E_0\cdot\psi
    \end{aligned}
  \]
\end{proof}

\begin{dfn}[Perturbation Matrix Constructor]{perturbation-matrix-constructor}
  Fix $\hat H'$. Let $\mathcal B(V_{E_0})$ denote the ordered orthonormal bases
  $A:\{1,\ldots,d\}\to V_{E_0}$ of $V_{E_0}$.
  The \emph{perturbation matrix constructor} is
  \[
    \pertmat:\mathcal B(V_{E_0})\to\mathbb C^{d\times d}
    \defeq A\map
    \left(\bra{A(r)}\hat H'\ket{A(s)}\right)_{r,s=1}^d.
  \]
  The input basis need not be a \xref[good basis]{def:good-basis}
\end{dfn}

\begin{dfn}[Perturbation Matrix]{perturbation-matrix}
  For $A\in\mathcal B(V_{E_0})$, its \emph{perturbation matrix} is the matrix
  generated by the \xref[perturbation matrix constructor]{def:perturbation-matrix-constructor}:
  \[
    W\defeq\pertmat(A)
  \]
\end{dfn}

\begin{thm}[Degenerate First-Order Equation]{degenerate-first-order-equation}
  Let $A(r)=\psi_r^{\porder{0}}$ be an ordered orthonormal basis of $V_{E_0}$,
  and let $W\defeq\pertmat(A)$.
  If $B(i)=\chi_i^{\porder{0}}$ is an ordered \xref[good basis]{def:good-basis}, then for each fixed $i$, with corresponding first-order energy correction $E_i^{\porder{1}}$, write
  \[
    \chi_i^{\porder{0}}
    = \sum_1^d\left(s\map c(s)\cdot\psi_s^{\porder{0}}\right),
    \qquad c : \{1,\ldots,d\}\to\mathbb C.
  \]
  Then its coefficient column satisfies
  \[
    W\mathbf c=E_i^{\porder{1}}\cdot\mathbf c,
    \qquad \mathbf c\ne0.
  \]
  In particular, giving the good basis $B$ to the constructor yields
  \[
    [\pertmat(B)]_{ri}=E_i^{\porder{1}}\cdot\delta_{r,i},
    \qquad r,i\in\{1,\ldots,d\}
  \]
\end{thm}

\begin{proof}
Given a degenerate eigenspace $V_{E_0}$ of dimension $d\geq2$, we have the unperturbed orthonormal degenerate eigenstates $\{\psi_s^{\porder{0}} : s\in\{1,\ldots,d\}\}$ with the same eigenvalue $E_0$

Given an orthonormal basis $A$, for the chosen good state $\chi_i^{\porder{0}}$, there exists a coefficient function $c : \{1,\ldots,d\}\to\mathbb C$ such that
\[\chi_i^{\porder{0}} = \sum_1^d\left(s \map c(s)\cdot\psi_s^{\porder{0}}\right)\] 

Where we get the original first order equation and multiply $\bra{\psi_r^{\porder{0}}}$ on both sides, we have
\[\bra{\psi_r^{\porder{0}}}\hat H_0\ket{\chi_i^{\porder{1}}} + \bra{\psi_r^{\porder{0}}}\hat H'\ket{\chi_i^{\porder{0}}} = \bra{\psi_r^{\porder{0}}}E_i^{\porder{0}}\ket{\chi_i^{\porder{1}}} + \bra{\psi_r^{\porder{0}}}E_i^{\porder{1}}\ket{\chi_i^{\porder{0}}}\] 

Notice that all states in $A$ have the same unperturbed energy $E_0$ as the degenerate eigenstates, we can obtain: 
\[\bra{\psi_r^{\porder{0}}}\hat H_0\ket{\chi_i^{\porder{1}}} = \braket{\hat H_0 \psi_r^{\porder{0}} \mid \chi_i^{\porder{1}}} = E_i^{\porder{0}} \braket{\psi_r^{\porder{0}} \mid \chi_i^{\porder{1}}} \] 

Plug in and extract the real number out, we have: 
\[E_i^{\porder{0}} \braket{\psi_r^{\porder{0}} \mid \chi_i^{\porder{1}}} + 
\bra{\psi_r^{\porder{0}}}\hat H'\ket{\chi_i^{\porder{0}}} = 
E_i^{\porder{0}} \braket{\psi_r^{\porder{0}} \mid \chi_i^{\porder{1}}} + 
E_i^{\porder{1}} \braket{\psi_r^{\porder{0}} \mid \chi_i^{\porder{0}}}\] 

Which means 
\[\bra{\psi_r^{\porder{0}}}\hat H'\ket{\chi_i^{\porder{0}}} = 
E_i^{\porder{1}} \braket{\psi_r^{\porder{0}} \mid \chi_i^{\porder{0}}}\] 

Plug in the expansion of $\chi_i^{\porder{0}}$, we have 
\[\bra{\psi_r^{\porder{0}}}\hat H'\ket{\sum_1^d\left(s \map c(s)\cdot\psi_s^{\porder{0}}\right)} = 
E_i^{\porder{1}} \braket{\psi_r^{\porder{0}} \mid \sum_1^d\left(s \map c(s)\cdot\psi_s^{\porder{0}}\right)}\] 

Due to linearity, 
\begin{align*}
\sum_1^d\left(s \map \bra{\psi_r^{\porder{0}}}\hat H'\ket{ c(s)\cdot\psi_s^{\porder{0}}} \right) &= 
E_i^{\porder{1}} \sum_1^d\left( s \map \braket{\psi_r^{\porder{0}} \mid c(s)\cdot\psi_s^{\porder{0}}}\right) \\ 
\sum_1^d\left(s \map c(s) \bra{\psi_r^{\porder{0}}}\hat H'\ket{ \psi_s^{\porder{0}}} \right) &= 
E_i^{\porder{1}} \sum_1^d\left( s \map c(s) \braket{\psi_r^{\porder{0}} \mid \psi_s^{\porder{0}}}\right)  
\end{align*}
Notice that $\braket{\psi_r^{\porder{0}} \mid \psi_s^{\porder{0}}} = \delta_{r,s}$, we have
\begin{align*}
\sum_1^d\left(s \map c(s) \bra{\psi_r^{\porder{0}}}\hat H'\ket{ \psi_s^{\porder{0}}} \right) &= 
E_i^{\porder{1}} c(r) 
\end{align*}

We treat each $c(r), r \in \{1, 2, ..., d\}$ as a variable, then we can assemble a linear system of equations, to solve $c$ pointwise. 
\begin{align*}
\begin{bmatrix}
\bra{\psi_1^{\porder{0}}}\hat H'\ket{\psi_1^{\porder{0}}} & \bra{\psi_1^{\porder{0}}}\hat H'\ket{\psi_2^{\porder{0}}} & \cdots & \bra{\psi_1^{\porder{0}}}\hat H'\ket{\psi_d^{\porder{0}}} \\
\bra{\psi_2^{\porder{0}}}\hat H'\ket{\psi_1^{\porder{0}}} & \bra{\psi_2^{\porder{0}}}\hat H'\ket{\psi_2^{\porder{0}}} & \cdots & \bra{\psi_2^{\porder{0}}}\hat H'\ket{\psi_d^{\porder{0}}} \\
\vdots & \vdots & \ddots & \vdots \\
\bra{\psi_d^{\porder{0}}}\hat H'\ket{\psi_1^{\porder{0}}} & \bra{\psi_d^{\porder{0}}}\hat H'\ket{\psi_2^{\porder{0}}} & \cdots & \bra{\psi_d^{\porder{0}}}\hat H'\ket{\psi_d^{\porder{0}}}
\end{bmatrix}
\begin{bmatrix}
c(1) \\
c(2) \\
\vdots \\
c(d)
\end{bmatrix}
=
E_i^{\porder{1}} \begin{bmatrix}
c(1) \\
c(2) \\
\vdots \\
c(d)
\end{bmatrix}
\end{align*}

According to the \xref[definition of the perturbation matrix]{def:perturbation-matrix}, let
\[
  W\defeq\pertmat(A),\qquad
  \mathbf c\defeq\begin{bmatrix}c(1)\\c(2)\\\vdots\\c(d)\end{bmatrix}.
\]
Then the linear system becomes
\[
  W\mathbf c=E_i^{\porder{1}}\cdot\mathbf c.
\]
Since $\chi_i^{\porder{0}}\ne0$, $\mathbf c\ne0$. Thus $E_i^{\porder{1}}$ is an eigenvalue of $W$, and $\mathbf c$ is a corresponding eigenvector.
Finding $\mathbf c$ is equivalent to finding $c$, since its $r$-th entry is $c(r)$

Giving the ordered good basis $B(r)=\chi_r^{\porder{0}}$ to the constructor
instead, the projected equation above gives
\[
  [\pertmat(B)]_{ri}
  =\bra{\chi_r^{\porder{0}}}\hat H'\ket{\chi_i^{\porder{0}}}
  =E_i^{\porder{1}}\cdot
    \braket{\chi_r^{\porder{0}}\mid\chi_i^{\porder{0}}}
  =E_i^{\porder{1}}\cdot\delta_{r,i}.
\]
Thus $W_B\defeq\pertmat(B)$ is diagonal. It is the matrix in basis $B$;
$W=\pertmat(A)$ is the matrix in basis $A$
\end{proof}

\begin{thm}[First-Order Converse]{degenerate-first-order-converse}
  Let $A$ and $W$ be as in the \xref[degenerate first-order equation]{thm:degenerate-first-order-equation}. Assume $V_{E_0}$ has a \xref[good basis]{def:good-basis} and $W$ has $d$ distinct eigenvalues. Given $c : \{1,\ldots,d\}\to\mathbb C$, let
  \[
    \chi^{\porder{0}}\defeq\sum_1^d\left(s\map c(s)\cdot\psi_s^{\porder{0}}\right),
    \qquad \sum_1^d\left(s\map\lvert c(s)\rvert^2\right)=1.
  \]
  If $W\mathbf c=e\cdot\mathbf c$, then $\chi^{\porder{0}}$ is a \xref[good state]{def:good-states}, with first-order energy correction $e$
\end{thm}

\begin{thm}[Constructing a Good Basis]{good-basis-from-perturbation-matrix}
  Choose an ordered orthonormal basis $A(s)=\psi_s^{\porder{0}}$ of $V_{E_0}$
  and calculate $W\defeq\pertmat(A)$.
  Assume $V_{E_0}$ has a \xref[good basis]{def:good-basis} and $W$ has $d$ distinct eigenvalues $e_1,\ldots,e_d$.
  Choose orthonormal eigenvectors $\mathbf c_1,\ldots,\mathbf c_d$ satisfying
  \[
    W\mathbf c_i=e_i\cdot\mathbf c_i,
    \qquad i\in\{1,\ldots,d\}.
  \]
  Write their entries as a function
  \[
    c : \{1,\ldots,d\}\times\{1,\ldots,d\}\to\mathbb C,
    \qquad \mathbf c_i=\begin{bmatrix}c(i,1)\\\vdots\\c(i,d)\end{bmatrix}.
  \]
  Then
  \[
    \chi_i^{\porder{0}}\defeq
    \sum_1^d\left(s\map c(i,s)\cdot\psi_s^{\porder{0}}\right),
    \qquad i\in\{1,\ldots,d\},
  \]
  form an ordered good basis $B(i)=\chi_i^{\porder{0}}$, with corresponding first-order energy corrections $E_i^{\porder{1}}=e_i$.
  Its perturbation matrix satisfies
  \[
    [\pertmat(B)]_{ij}=e_i\cdot\delta_{i,j},
    \qquad i,j\in\{1,\ldots,d\}
  \]
\end{thm}

\begin{proof}
  Each $\chi_i^{\porder{0}}$ is a good state with first-order energy correction $e_i$ by the \xref[first-order converse]{thm:degenerate-first-order-converse}.
  Since $A$ and the coefficient vectors are orthonormal,
  \[
    \braket{\chi_i^{\porder{0}}\mid\chi_j^{\porder{0}}}
    =\sum_1^d\left(s\map c(i,s)^*\cdot c(j,s)\right)
    =\delta_{i,j}.
  \]
  Thus the $d$ states form an orthonormal basis of the $d$-dimensional space $V_{E_0}$, hence a good basis.
  The \xref[degenerate first-order equation]{thm:degenerate-first-order-equation}
  applied in this good basis shows that $\pertmat(B)$ has diagonal entries $e_i$
\end{proof}
