\section{Introduction to Symmetry}

\begin{dfn}[Symmetry]{symmetry}
  A \emph{symmetry} of a system is some transformation, such that the system is invariant before and after the transformation.
\end{dfn}

\begin{dfn}[Symmetry of Quantum System]{symmetry-qm}
  A \emph{symmetry} of a quantum system is a transformation that leaves the Hamiltonian invariant.
\end{dfn}

\begin{dfn}[Types of Symmetry]{symmetry-types}
  There are several types of symmetries in quantum mechanics, including:
  \begin{itemize}[nosep]
    \item \emph{Translation symmetry}: The system is invariant under some spatial translations.
    \item \emph{Rotation symmetry}: The system is invariant under some rotations.
    \item \emph{Reflection symmetry}: The system is invariant under reflections on some axis/planes.
    \item \emph{Inversion symmetry}: The system is invariant under inversion of all coordinates.
    \item \emph{Time-translation symmetry}: The system is invariant under time shifts.
  \end{itemize}
\end{dfn}

\begin{thm}[Symmetry Condition on the Hamiltonian]{symmetry-condition-hamiltonian}
  $\hat{U}$ is a \xref{def:symmetry-qm} of $\hat{H}$ iff
  \[
    \hat{U}^{\dagger}\hat{H}\hat{U} = \hat{H}.
  \]
\end{thm}

\begin{cor}[Commutator Form]{commutator-form-symmetry}
  For unitary $\hat{U}$,
  \[
    \hat{U}^{\dagger}\hat{H}\hat{U} = \hat{H} \iff \comm{\hat{U}}{\hat{H}} = 0.
  \]
\end{cor}

\begin{proof}
  Left-multiply by $\hat{U}$ and use $\hat{U}\hat{U}^{\dagger} = \hat{I}$:
  \[
    \hat{U}^{\dagger}\hat{H}\hat{U} = \hat{H} \iff \hat{H}\hat{U} = \hat{U}\hat{H} \iff \comm{\hat{U}}{\hat{H}} = 0.
  \]
\end{proof}

\begin{rem}[``$\hat{A}$ Obtained After Applying $\hat{U}$'']{conjugation-vocabulary}
  The phrase ``$\hat{A}$ obtained after applying $\hat{U}$'' refers to the \xref[conjugation]{ch03::def:conjugation-action} $\hat{U}^{\dagger}\hat{A}\hat{U}$, since this preserves the expectation value.
\end{rem}

\section{Groups}

\begin{dfn}[Group]{group}
  A \emph{group} consists of the following data and assumptions:
  \begin{itemize}[nosep]
    \item A set $G$.
    \item A binary operation $\cdot : G \times G \to G$ (the codomain being $G$ bakes in closure).
    \item An identity element $e \in G$.
    \item An inverse function $(\cdot)^{-1} : G \to G$, written $g \mapsto g^{-1}$.
    \item \emph{Associativity}: $(g_1 \cdot g_2) \cdot g_3 = g_1 \cdot (g_2 \cdot g_3)$ for all $g_1, g_2, g_3 \in G$.
    \item \emph{Identity axiom}: $e \cdot g = g \cdot e = g$ for all $g \in G$.
    \item \emph{Inverse axiom}: $g \cdot g^{-1} = g^{-1} \cdot g = e$ for all $g \in G$.
  \end{itemize}
  A collection with a well-behaved multiplication and inverses.
\end{dfn}

\begin{ex}[Real Numbers Under Addition]{reals-group}
  Choose:
  \begin{itemize}[nosep]
    \item Set $G := \mathbb{R}$ (the real numbers).
    \item Operation $\cdot := +$ (ordinary addition).
    \item Identity $e := 0$.
    \item Inverse $a^{-1} := -a$.
  \end{itemize}
  Associativity, identity, and inverse axioms follow from the usual properties of addition.
\end{ex}

\begin{ex}[Boolean Set Under XOR]{boolean-group}
  Choose:
  \begin{itemize}[nosep]
    \item Set $G := \{\text{tt}, \text{ff}\}$ (equivalently $\{0, 1\}$).
    \item Operation $\cdot := \oplus$ (XOR, equivalent to addition mod $2$).
    \item Identity $e := \text{ff}$ (equivalently $0$).
    \item Inverse $g^{-1} := g$ (each element is self-inverse).
  \end{itemize}
  All axioms hold by direct verification.
\end{ex}

\begin{dfn}[Continous Group]{continuous-group}
  A group is \emph{continuous} (a \emph{Lie group}) if its elements are parametrized by a {\color{blue}continuous parameter space}, and multiplication and inversion vary smoothly with the parameters. The \xref[real numbers under addition]{ex:reals-group} is an example.
\end{dfn}

\begin{dfn}[Discrete Group]{discrete-group}
  A group is \emph{discrete} if its elements form a {\color{blue}discrete set} (finite or {\color{blue}countable}), with no continuous parameter connecting distinct elements. The \xref[Boolean set under XOR]{ex:boolean-group} is an example.
\end{dfn}

% \begin{rem}[Why the Distinction Matters]{continuous-discrete-why}

Discrete groups have no such infinitesimal structure, so a discrete symmetry is defined directly by its operator rather than by a generator.

\begin{dfn}[Continuous Symmetry]{continuous-symmetry}
  A symmetry is \emph{continuous} if its operators form a \xref[continuous group]{def:continuous-group}.
\end{dfn}

\begin{dfn}[Discrete Symmetry]{discrete-symmetry}
  A symmetry is \emph{discrete} if its operators form a \xref[discrete group]{def:discrete-group}.
\end{dfn}

\section{Translation Operator}

\begin{dfn}[Translation Operator]{translation-operator}
  Intuitively, a \emph{translation operator} $\hat{T}(\vec{r}_0)$ (translates a state by a vector $\vec{r}_0$) acts on a state $\ket{\psi}$ should let $\ket{\psi}$ be translated by a vector $\vec{r}$, i.e.
  \[\hat{T}(\vec{r}_0)\ket{\psi} = \ket{\psi(\vec{r} - \vec{r}_0)}\]
\end{dfn}

\begin{dfn}[Translation Operator (1D)]{translation-operator-1d}
  In one dimension, the translation operator $\hat{T}(a)$ shifts a state by a real distance $a \in \mathbb{R}$. Its action on a position-space wavefunction is
  \[
    \hat{T}(a)\,\psi(x) = \psi(x - a).
  \]
\end{dfn}

\begin{thm}[Translation Operators Form a Symmetry Group]{translation-symmetry-group}
  Suppose $\hat{T}(a)$ is a \xref[symmetry operator]{def:symmetry-qm} of a Hamiltonian $\hat{H}$ for every $a \in \mathbb{R}$. Then the collection of translation operators forms a group under operator composition:
  \begin{itemize}[nosep]
    \item Set $G := \{\hat{T}(a) : a \in \mathbb{R}\}$.
    \item Operation $\cdot$: operator composition, $\hat{T}(a) \cdot \hat{T}(b) = \hat{T}(a+b)$.
    \item Identity $e := \hat{T}(0) = \hat{I}$.
    \item Inverse $\hat{T}(a)^{-1} := \hat{T}(-a)$.
  \end{itemize}
  All axioms follow from addition on $\mathbb{R}$.
\end{thm}

\begin{thm}[Exponential Form of the Translation Operator]{translation-exponential}
  The one-dimensional translation operator can be written as:
  \[\hat{T}(a) = e^{-i a \hat{p}/\hbar}\]
\end{thm}

\begin{proof}
  Expand the exponential in a \xref[Taylor series]{ch03::def:operator-function} at $0$ 
  \begin{align*}
  \hat{T}(a) &= e^{-i a \hat{p}/\hbar} = \sum_{n=0}^{\infty} \frac{(-ia/\hbar)^n}{n!} \cdot \hat{p}^n = \sum_{n=0}^{\infty} \frac{(-i a \hat{p})^n}{n! \hbar^{n}}
  \end{align*} 

  According to the \xref[definition of the momentum operator]{ch02::def:momentum-operator-position}, $\hat{p} = -i\hbar \frac{d}{dx}$, so we can substitute this into the series expansion:
  \begin{align*}
  \hat{T}(a) &= \sum_{n=0}^{\infty} \frac{(-i a (-i\hbar \frac{d}{dx}))^n}{n! \hbar^{n}} = \sum_{n=0}^{\infty} \frac{(- a ( \frac{d}{dx}))^n}{n!} 
  \end{align*}

  So, 
  \[\hat{T}(a) \psi(x) = \sum_{n=0}^{\infty} \frac{(- a ( \frac{d}{dx}))^n}{n!} \psi(x) = \sum_{n=0}^{\infty} \frac{(- a)^n}{n!} \psi^{(n)}(x)\] 
  
  Then, we do a Taylor expansion of $\psi(x)$ centered at $x + a$: 
  \[\psi(x) = \sum_{n=0}^{\infty} \frac{\psi^{(n)}(x + a)}{n!} (x - (x + a))^n = \sum_{n=0}^{\infty} \frac{\psi^{(n)}(x + a)}{n!} (-a)^n = \sum_{n=0}^{\infty} \frac{(-a)^n}{n!} \psi^{(n)}(x + a) \] 

  Substituting $x \to x - a$ in the previous identity,
  \[\psi(x - a) = \sum_{n=0}^{\infty} \frac{(-a)^n}{n!} \psi^{(n)}((x - a) + a) = \sum_{n=0}^{\infty} \frac{(-a)^n}{n!} \psi^{(n)}(x).\]

  Which means, 
  \[\hat{T}(a) \psi(x) = \psi(x - a)\] 
\end{proof}

\begin{thm}[Position Operator After Applying Translation]{position-under-translation}
  Applying $\hat{T}(a)$ to the position operator $\hat{x}$ (see \xref{rem:conjugation-vocabulary}) gives
  \[
    \hat{T}(a)^{\dagger}\,\hat{x}\,\hat{T}(a) \;=\; \hat{x} + a\hat{I}.
  \]
\end{thm}

\begin{proof}
  Define $\hat{x}' := \hat{T}(a)^{\dagger}\,\hat{x}\,\hat{T}(a)$, and consider its action on a wavefunction $\psi(x)$: 
  \[\hat{x}' \psi(x) = \hat{T}(a)^{\dagger}\,\hat{x}\,\hat{T}(a) \, \psi(x)\]

  Since operators are associative, we have 
  \[\hat{x}' \psi(x) = \hat{T}(a)^{\dagger}\,\hat{x}\,(\hat{T}(a) \, \psi(x)) = \hat{T}(a)^{\dagger}\,(\hat{x}\, \psi(x-a)) = \hat{T}(a)^{\dagger}\,(x \, \psi(x-a))\]

  Treat $g(x) = \hat{x}\, \psi(x-a)$, so, $T(a) \, g(x) = g(x) = \hat{x-a}\, \psi((x-a)-a)$. 

  Notice that $T(a)$ is a unitary operator, ie $T(a)^\dagger = T(a)^{-1} = T(-a)$, we have 
  \[\hat{x}' \psi(x) = \hat{T}(-a)\,(x \, \psi(x-a)) = (x+a) \, \psi((x+a)-a) = (x+a) \, \psi(x)\]

  Rewrite the right-hand side in operator form:
  \[(x+a)\,\psi(x) = x\,\psi(x) + a\,\psi(x) = \hat{x}\,\psi(x) + a\hat{I}\,\psi(x) = (\hat{x} + a\hat{I})\,\psi(x).\]

  So $\hat{x}'\,\psi = (\hat{x} + a\hat{I})\,\psi$ for every $\psi$. Since two operators agreeing on all inputs are equal, we conclude
  \[\hat{x}' = \hat{x} + a\hat{I}.\]
\end{proof}

\begin{thm}[Momentum Operator After Applying Translation]{momentum-under-translation}
  Applying $\hat{T}(a)$ to the momentum operator $\hat{p}$ (see \xref{rem:conjugation-vocabulary}) gives
  \[
    \hat{T}(a)^{\dagger}\,\hat{p}\,\hat{T}(a) \;=\; \hat{p}.
  \]
\end{thm}

\begin{proof}
  Define $\hat{p}' := \hat{T}(a)^{\dagger}\,\hat{p}\,\hat{T}(a)$, and consider its action on a wavefunction $\psi(x)$: 
  \[\hat{p}' \psi(x) = \hat{T}(a)^{\dagger}\,\hat{p}\,\hat{T}(a) \, \psi(x) = \hat{T}(a)^{\dagger}\,\hat{p}\,\psi(x-a)\]

  According to the definition of \xref{ch02::def:momentum-operator-position}, $\hat p = -i \hbar d/dx $, plug in:
  \[\hat{p}' \psi(x) = - \hat{T}(a)^{\dagger}\,i \hbar \, \frac{d\psi(x-a)}{dx}\]

  Use $\hat{T}(a)^{\dagger} = \hat T(-a)$,
  \[\hat{p}' \psi(x) = - \hat{T}(-a)\,i \hbar \, \frac{d\psi(x-a)}{dx} = - i \hbar \, \frac{d\psi(x)}{dx} = \hat p \psi(x)\]

  So $\hat{p}'\,\psi = \hat{p}\,\psi$ for every $\psi$. Since two operators agreeing on all inputs are equal, we conclude
  \[\hat{p}' = \hat{p}.\]
\end{proof}

\begin{thm}
For all operator, which only consists of $\hat x$ and $\hat p$, the conjugation action of $\hat T(a)$ on the operator is equivalent to replacing every $\hat x$ with $\hat x + a\hat I$ and leaving every $\hat p$ unchanged. ie,
\[Q(\hat x, \hat p) \longmapsto \hat T(a)^\dagger Q(\hat x, \hat p) \hat T(a) = Q(\hat x + a\hat I, \hat p)\]
\end{thm}

\begin{thm}[Translation Symmetry Condition on the Potential]{translation-condition-potential}
  Let $\hat{H}$ be a \xref[1D Hamiltonian]{ch02::def:hamiltonian-1d}. If $\hat{T}(a)$ is a \xref[symmetry operator]{def:symmetry-qm} of $\hat{H}$, then the potential is invariant under translation by $a$:
  \[
    V(x + a) = V(x) \qquad \text{for all } x.
  \]
\end{thm}

\begin{proof}
  Since $\hat T(a)$ is a unitary symmetry, \xref{cor:commutator-form-symmetry} gives
  \[\hat T(a)^{\dagger} \hat H \hat T(a) = \hat H \iff \comm{\hat H}{\hat T(a)} = 0.\]

  Notice that $\hat H = \frac{\hat p^2}{2m} + V(\hat x)$, and by \xref{thm:momentum-under-translation} and \xref{thm:position-under-translation}, we have
  \[\hat T(a)^{\dagger} \hat H \hat T(a) = \hat T(a)^{\dagger} \left(\frac{\hat p^2}{2m} + V(\hat x)\right) \hat T(a) = \frac{\hat p^2}{2m} + V(\hat x + a\hat I).\] 

  Which means, $\hat T(a)$ is a symmetry of $\hat H$ iff
  \[V(x) = V(x+a)\]
\end{proof}

\begin{thm}[Eigenvalues of the Translation Operator]{translation-eigenvalues}
  The eigenvalues of the translation operator $\hat{T}(a)$ have modulus $1$:
  \[
    \hat{T}(a)\ket{\psi} = \lambda\ket{\psi} \implies \lambda = e^{i\theta}
    \quad \text{for some } \theta \in [0, 2\pi).
  \]
\end{thm}

\begin{proof}
  Since $\hat{T}(a)$ is unitary (\xref{thm:translation-unitary}),
  \[\braket{\hat{T}(a)\,\psi \mid \hat{T}(a)\,\psi} = \braket{\psi \mid \hat{T}(a)^\dagger \hat{T}(a) \mid \psi} = \braket{\psi \mid \psi}.\]
  On the other hand,
  \[\braket{\hat{T}(a)\,\psi \mid \hat{T}(a)\,\psi} = \braket{\lambda\psi \mid \lambda\psi} = |\lambda|^2\braket{\psi \mid \psi}.\]
  Equating and dividing by $\braket{\psi \mid \psi} \neq 0$:
  \[|\lambda|^2 = 1 \implies \lambda = e^{i\theta} \quad\text{for some } \theta \in [0, 2\pi).\]
\end{proof}

\begin{thm}[Bloch's Theorem]{bloch-theorem}
  If $T(a)$ is a symmetry of $\hat H$ for some $a \in \mathbb{R}$, stationary states of $\hat H$ have the form
  \[\psi(x) = e^{iqx} u(x)\]

  Where 
  \begin{itemize}[nosep]
    \item $q$ is called the crystal momentum 
    \item $u(x)$ is periodic and have a period of $a$ 
  \end{itemize} 
\end{thm} 

\begin{rem}[Stationary States may not be Periodic]
From \xref{thm:bloch-theorem}, we can see that the stationary states $\psi(x)$ is not periodic. However $u(x)$ is periodic with period $a$. 
\end{rem}

\begin{proof}
Since $\hat T(a)$ is a symmetry of $\hat H$, we have $\comm{\hat H}{\hat T(a)} = 0$. According to \xref{ch03::thm:commuting-share-eigenvectors}, stationary states of $\hat H$ can be chosen as simultaneous eigenvectors of $\hat T(a)$: $\hat T(a) \ket{\psi} = \lambda \ket{\psi}$.

By \xref{thm:translation-eigenvalues}, $\lambda = e^{i\theta}$ for some $\theta \in [0, 2\pi)$. Since $\hat T(a)$ may be written as $e^{-iap/\hbar}$, we identify $\theta = qa$ where $q = -p/\hbar$.

Then, we have
\[\hat T(a) \ket{\psi} = e^{iqa} \ket{\psi} \implies \psi(x-a) = e^{iqa} \psi(x)\] 

Change $a \to -a$ to rewrite in the standard form:
\[\psi(x+a) = e^{iqa}\,\psi(x).\]

Define $u(x) := e^{-iqx}\,\psi(x)$. Then
\[
  u(x+a) = e^{-iq(x+a)}\,\psi(x+a) = e^{-iqx}\,e^{-iqa}\,e^{iqa}\,\psi(x) = u(x),
\]
where $e^{-iqa}\,e^{iqa} = 1$. So $u$ has period $a$, and rearranging gives $\psi(x) = e^{iqx}\,u(x)$.

Although $\hat{H}$ is translation-invariant, the state $\psi$ is \emph{not}:
\[\psi(x+a) = e^{iqa}\,\psi(x) \neq \psi(x).\]

However, the observable $|\psi|^2$ still respects the symmetry, since $|e^{iqa}| = 1$:
\[|\psi(x+a)|^2 = |e^{iqa}|^2\,|\psi(x)|^2 = |\psi(x)|^2.\]
\end{proof}

\begin{thm}[Continuous Translation Symmetry Forces Constant $V$]{continuous-translation-constant-v}
  If $\hat{T}(a)$ is a symmetry of $\hat{H}$ for every $a \in \mathbb{R}$, then $V$ is constant.
\end{thm}

\section{Rotational Symmetry in 3D}

\begin{dfn}[Rotation Operator]{rotation-operator}
  The \emph{rotation operator} $\hat{R}_{\hat{z}}(\varphi)\psi$ rotates a state $\psi$ about the $z$-axis by an angle $\varphi$:
  \[\hat{R}_{\hat{z}}(\varphi)\psi(r, \theta, \phi) = \psi(r, \theta, \phi - \varphi)\] 
\end{dfn}

\begin{rem}
  We may also define rotation operators about the $x$-axis and $y$-axis, $\hat{R}_{\hat{x}}(\varphi)$ and $\hat{R}_{\hat{y}}(\varphi)$, in a similar manner. 
\end{rem}

\begin{thm}[Exponential Form of the Rotational Operator]{rotation-exponential}
  The rotation operator about the $z$-axis can be written as:
  \[\hat{R}_z(\varphi) = \exp\!\left[-\frac{i\varphi}{\hbar}\,\hat{L}_z\right]\] 
\end{thm} 

\begin{proof}
  Do taylor's expansion on the exponential function:
  \[\hat{R}_z(\varphi) = \sum_{n=0}^{\infty} \frac{1}{n!} \left(-\frac{i\varphi}{\hbar}\,\hat{L}_z\right)^n\]

  According to \xref{ch04::def:angular-momentum}, the angular momentum operator about the $z$-axis is defined as
  \[\hat{L}_z = \hat x \hat p_y - \hat y \hat p_x\]. 

  Which means that $\hat{R}_z(\varphi) \psi(r, \theta, \phi)$ is 
\[\hat{R}_z(\varphi) \psi(r, \theta, \phi) = \sum_{n=0}^{\infty} \frac{1}{n!} \left(-\frac{i\varphi}{\hbar}\,(\hat x \hat p_y - \hat y \hat p_x)\right)^n \psi(r, \theta, \phi)\] 

  Plug in the definition of $\hat p_x$ and $\hat p_y$ from \xref{ch02::def:momentum-operator-position}, we have
  \begin{align*}
    \hat{R}_z(\varphi) \psi(r, \theta, \phi) &= \sum_{n=0}^{\infty} \frac{1}{n!} \left(-\frac{i\varphi}{\hbar}\bigl(\hat x (-i\hbar \tfrac{\partial}{\partial y}) - \hat y (-i\hbar \tfrac{\partial}{\partial x})\bigr)\right)^n \psi(r, \theta, \phi)\\
    &= \sum_{n=0}^{\infty} \frac{1}{n!} \bigl(-\varphi\,(\hat x \tfrac{\partial}{\partial y} - \hat y \tfrac{\partial}{\partial x})\bigr)^n \psi(r, \theta, \phi).
  \end{align*}

  {\color{blue}In cylindrical/spherical coordinates the chain rule gives $\hat x \tfrac{\partial}{\partial y} - \hat y \tfrac{\partial}{\partial x} = \tfrac{\partial}{\partial \phi}$ (multivariable calculus prerequisite, TODO: prove separately),} so
  \[
    \hat{R}_z(\varphi) \psi(r, \theta, \phi) = \sum_{n=0}^{\infty} \frac{(-\varphi)^n}{n!}\,\frac{\partial^n \psi}{\partial \phi^n}(r, \theta, \phi).
  \]

  This is the Taylor expansion of $\psi(r, \theta, \phi - \varphi)$ around $\phi$:
  \[
    \psi(r, \theta, \phi - \varphi) = \sum_{n=0}^{\infty} \frac{(-\varphi)^n}{n!}\,\frac{\partial^n \psi}{\partial \phi^n}(r, \theta, \phi).
  \]

  Therefore $\hat{R}_z(\varphi)\,\psi(r,\theta,\phi) = \psi(r,\theta,\phi-\varphi)$, matching \xref{def:rotation-operator}.
\end{proof}

\section{Parity/Inverse Symmetry}

\begin{dfn}
[Parity Operator]{parity-operator}
  The \emph{parity operator} $\Pi$ (also called the \emph{inversion operator}) is defined by
  \[ \Pi \ket{\psi(\vec r)} = \ket{\psi(-\vec r)} \]
\end{dfn}

\begin{thm}[Parity Squared is Identity]{parity-squared}
  $\Pi^2 = \hat{I}$.
\end{thm}

\begin{proof}
  Applying $\Pi$ twice to any $\psi$:
  \[\Pi^2\,\psi(\vec r) = \Pi\,\psi(-\vec r) = \psi(-(-\vec r)) = \psi(\vec r).\]
  Since this holds for every $\psi$, $\Pi^2 = \hat{I}$.
\end{proof}

\begin{thm}[Eigenvalues of the Parity Operator]{parity-eigenvalues}
  The eigenvalues of the parity operator $\Pi$ are $\pm 1$.
\end{thm}

\begin{proof}
  If $\Pi\ket{\lambda} = \lambda\ket{\lambda}$, then $\Pi^2\ket{\lambda} = \lambda^2\ket{\lambda}$. By \xref{thm:parity-squared}, $\Pi^2 = \hat{I}$, so $\lambda^2 = 1$, hence $\lambda = \pm 1$.
\end{proof}

\begin{thm}[Parity Operator is Hermitian]{parity-hermitian}
  The parity operator $\Pi$ is Hermitian, i.e. $\Pi^{\dagger} = \Pi$.
\end{thm}

\begin{proof}
  For any state $\ket{f}$, $\ket{g}$, we have
  \[\braket{f \mid \Pi g} = \int_{-\infty}^{\infty} f^*(x) \Pi g(x) \, dx = \int_{-\infty}^{\infty} f^*(x) g(-x) \, dx = \int_{-\infty}^{\infty} (f(-x))^* g(x) \, dx = \braket{\Pi f \mid g}\]

  Notice that $\braket{f \mid \Pi g} = \braket{\Pi f \mid g} \implies \Pi = \Pi^{\dagger}$. So $\Pi$ is Hermitian. 
\end{proof}

\begin{thm}[Parity Operator is Unitary]{parity-unitary}
  The parity operator $\Pi$ is unitary, i.e. $\Pi^{\dagger} = \Pi^{-1}$.
\end{thm}

\begin{proof}
  Since $\Pi^{\dagger} = \Pi$, and according to \xref{thm:parity-squared}, $\Pi \Pi = \Pi \Pi^{\dagger} = \hat{I}$, so $\Pi^{\dagger} = \Pi^{-1}$, and $\Pi$ is unitary. 
\end{proof}

\begin{thm}[Parity Operator is Discrete Symmetry]{parity-discrete}
  The parity operator $\Pi$ is a discrete symmetry. 
\end{thm} 

\begin{proof}
  The parity operators form a group:
  \begin{itemize}[nosep]
    \item Set $G := \{\hat{I}, \Pi\}$.
    \item Operation $\cdot$: operator composition.
    \item Identity $e := \hat{I}$.
    \item Inverse $\Pi^{-1} := \Pi$ (by \xref{thm:parity-squared}, parity is self-inverse).
  \end{itemize}
  This is a \xref[discrete group]{def:discrete-group} with two elements, so parity is a \xref{def:discrete-symmetry}.
\end{proof}

\begin{thm}[Position Operator After Applying Parity]{position-under-parity}
  Applying $\Pi$ to the position operator $\hat{x}$ (see \xref{rem:conjugation-vocabulary}) gives
  \[
    \Pi^{\dagger}\,\hat{x}\,\Pi = -\hat{x}.
  \]
\end{thm}

\begin{proof}
  Consider $\Pi^{\dagger}\hat{x}\Pi$ acting on a test function $f(x)$:
  \[\Pi^{\dagger}\hat{x}\Pi\, f(x) = \Pi^{\dagger}\hat{x}\,f(-x) = \Pi^{\dagger}\bigl(x\,f(-x)\bigr).\]
  Since $\Pi$ is \xref[hermitian]{thm:parity-hermitian} and \xref[unitary]{thm:parity-unitary}, we have $\Pi^{\dagger} = \Pi$. Define $g(x) := x\,f(-x)$; then
  \[\Pi^{\dagger}\hat{x}\Pi\, f(x) = \Pi\, g(x) = g(-x) = -x\,f(x).\]
  So $\Pi^{\dagger}\hat{x}\Pi\, f = -\hat{x}\, f$ for every $f$, giving $\Pi^{\dagger}\hat{x}\Pi = -\hat{x}$.
\end{proof}

\begin{thm}[Momentum Operator After Applying Parity]{momentum-under-parity}
  Applying $\Pi$ to the momentum operator $\hat{p}$ gives
  \[
    \Pi^{\dagger}\,\hat{p}\,\Pi = -\hat{p}.
  \]
\end{thm}

\begin{proof}
  Consider $\Pi^{\dagger}\hat{p}\Pi$ acting on a test function $f(x)$:
  \[\Pi^{\dagger}\hat{p}\Pi\, f(x) = \Pi^{\dagger}\hat{p}\,f(-x) = \Pi^{\dagger}\left(-i\hbar\,\frac{d}{dx}f(-x)\right).\]
  Since $\Pi$ is \xref[hermitian]{thm:parity-hermitian} and \xref[unitary]{thm:parity-unitary}, we have $\Pi^{\dagger} = \Pi$. Define $g(x) := -i\hbar\,\frac{d}{dx}f(-x)$; then
  \[\Pi^{\dagger}\hat{p}\Pi\, f(x) = \Pi\, g(x) = g(-x) = i\hbar\,\frac{d}{dx}f(x).\]
  So $\Pi^{\dagger}\hat{p}\Pi\, f = -\hat{p}\, f$ for every $f$, giving $\Pi^{\dagger}\hat{p}\Pi = -\hat{p}$.
\end{proof}

\begin{thm}[General Operator After Applying Parity]{operator-under-parity}
  For any operator $Q(\hat{x}, \hat{p})$ built from $\hat{x}$ and $\hat{p}$,
  \[
    \Pi^{\dagger}\,Q(\hat{x}, \hat{p})\,\Pi = Q(-\hat{x}, -\hat{p}).
  \]
\end{thm}

\begin{thm}[Angular Momentum is Parity-Invariant]{angular-momentum-under-parity}
  $\Pi^{\dagger}\,\hat{\vec{L}}\,\Pi = \hat{\vec{L}}$.
\end{thm}

\begin{proof}

  \]
\end{proof}

\section{Pictures}

\begin{dfn}[Pictures]{pictures}
  In quantum mechanics, a \emph{picture} is a convention for how to distribute the time dependence between states and 
  operators
\end{dfn} 

\begin{dfn}[Schr\"odinger Picture]{schrodinger-picture}
  The \emph{Schr\"odinger picture} keeps the state dependent on time, while the operators are time-independent. ie, 
  \[\text{state:} \, \ket{S(t)}, \, \text{expected value:} \, \braket{Q} = \braket{S(t) | Q | S(t)}\]
\end{dfn}

\begin{dfn}[Heisenberg Picture]{heisenberg-picture}
  The \emph{Heisenberg picture} keeps the state independent of time. Instead, the time dependence is transferred to the operators. 
  \[
    \text{state:} \, \ket{H}, \, \text{expected value:} \, \braket{Q_H(t)} = \braket{H | Q_H(t) | H}
  \]
  where $U(t)$ is the \xref[time-translation operator]{def:time-translation-operator}. 
\end{dfn}

\begin{rem}[Pictures Are Conjugation Actions]{pictures-are-conjugation}
  The Heisenberg operator $Q_H(t) = U^{\dagger}(t)\,Q\,U(t)$ is the \xref[conjugation action]{ch03::def:conjugation-action} of $U(t)$ on $Q$. Expectation values agree between the Schr\"odinger and Heisenberg pictures because that is exactly what the conjugation action preserves.
\end{rem}

\begin{ex}
Consider $H = \frac{p^2}{2m} + V(x)$, find in the Heisenberg picture, the $x_H{\delta}$ operator where $\delta$ is infinitesimal. 

Since $\delta$ is small, we may do taylor's expansion and only keep the first order term (at t = 0), and we know that $e^{it \hat H}{\hbar}' = \frac{i \hat H}{\hbar}$

\[x_H(\delta) = \hat U^{\dagger}(\delta) \hat x \hat U(\delta) \approx \left( 1 + \frac{i}{\hbar} \delta \hat H \right) \hat x \left( 1 - \frac{i}{\hbar} \delta \hat H \right) \]

Expand and omitting the second order term, we have
\[x_H(\delta) \approx \hat x + \frac{i}{\hbar} \comm{\hat H}{\hat x} \delta\]

Apply \xref[Hamiltonian Commutators]{ch03::thm:hamiltonian-commutators} to substitute $\comm{\hat H}{\hat x} = -\dfrac{i\hbar}{m}\hat p$:
\[
  x_H(\delta) \approx \hat x + \frac{i}{\hbar}\cdot\left(-\frac{i\hbar}{m}\hat p\right)\delta = \hat x + \frac{\hat p}{m}\,\delta.
\]
\end{ex}

\begin{rem}
This matches the classical equation of motion $v = p/m$: the infinitesimal time-evolution of $\hat x$ in the Heisenberg picture reproduces the classical relation of $x = x_0 + vt$.
\end{rem}

\begin{thm}[Heisenberg Equation of Motion]{heisenberg-equation}
  Assuming $\hat{Q}$ and $\hat{H}$ are independent of time 
  \[
    \frac{d}{dt}\hat{Q}_H(t) \;=\; \frac{1}{i\hbar}\,\comm{\hat{Q}_H(t)}{\hat{H}}.
  \]
\end{thm}

\begin{proof}
  By the definition of the Heisenberg picture, $\hat{Q}_H(t) = U^{\dagger}(t)\,\hat{Q}\,U(t)$. 

  From the \xref[Generalized Ehrenfest's Theorem]{ch03::thm:ehrenfest}, we have 
  \[
    \frac{d}{dt}\hat{Q}_H(t) = \frac{1}{i\hbar}\,\comm{U^{\dagger}(t)\, \hat{Q}\,U(t)}{\hat{H}} + \frac{\partial U^{\dagger}(t)\, \hat{Q}\,U(t)}{\partial t} 
  \] 

  Since $\frac{d}{dt}U(t) = \frac{i\hat{H}}{\hbar}\,U(t)$, then apply the product rule to the second term:
  \begin{align*}
  \frac{d}{dt}\hat{Q}_H(t) &= \frac{1}{i\hbar}\,\comm{U^{\dagger}(t)\, \hat{Q}\,U(t)}{\hat{H}} + \frac{\partial U^{\dagger}(t)}{\partial t} \hat{Q}\,U(t) + U^{\dagger}(t) \frac{\partial \hat{Q} }{\partial t} U(t) + U^{\dagger}(t) \hat{Q} \frac{\partial U(t)}{\partial t} \\
    &= \frac{1}{i\hbar}\,\comm{U^{\dagger}(t)\, \hat{Q}\,U(t)}{\hat{H}} + \frac{-i\hat{H}}{\hbar} U^{\dagger}(t) \hat{Q} U(t) + U^{\dagger}(t) \hat{Q} \frac{i\hat{H}}{\hbar} U(t)
  \end{align*}

  Since if $\hat{Q}$ is independent of time, then $\frac{\partial \hat{Q} }{\partial t} = 0$.

  Applying \xref{lem:conjugation-preserves-commutator} with $\hat{A} = U(t)$, $\hat{X} = \hat{Q}$, $\hat{C} = \hat{H}$ (since $\comm{U(t)}{\hat{H}} = 0$ by \xref{thm:time-translation-commutes-with-h}), we can simplify
  \[
    \comm{U^{\dagger}(t)\,\hat{Q}\,U(t)}{\hat{H}}
    \;=\; U^{\dagger}(t)\,\comm{\hat{Q}}{\hat{H}}\,U(t)
    \;=\; \comm{\hat{Q}_H(t)}{\hat{H}}.
  \]
\end{proof}
